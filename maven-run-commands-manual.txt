#To run failed testcases (both work the same way with no difference)
mvn test -Dtest=FailedRun -Dcucumber.features="@failedrerun.txt"
mvn test -Dcucumber.features="@failedrerun.txt" -Dtest=FailedRun

#To run specific feature
mvn test -Dcucumber.features=src/test/java/com/qa/features/Elements.feature

#To run specific scenarios containing specific tags in a specific feature 
mvn test -Dcucumber.feature="src/test/java/com/qa/features/Widgets.feature" -Dcucumber.filter.tags="@TestOne"
